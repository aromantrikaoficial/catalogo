    function filterCategory(cat, element) {
      document.querySelectorAll('.cat-btn').forEach(btn => btn.classList.remove('active'));
      if (element) element.classList.add('active');
      renderProducts(cat);
    }

    function addToCart(id) {
      const item = cart.find(p => p.id === id);
      if (item) {
        item.qty++;
      } else {
        const prod = products.find(p => p.id === id);
        if (prod) cart.push({ ...prod, qty: 1 });
      }
      updateCartUI();
      toggleCart(true);
    }

    function updateCartUI() {
      const cartItems = document.getElementById('cart-items');
      const cartCount = document.getElementById('cart-count');
      const cartTotal = document.getElementById('cart-total');

      const totalCount = cart.reduce((acc, p) => acc + p.qty, 0);
      const totalPrice = cart.reduce((acc, p) => acc + (p.price * p.qty), 0);

      if (cartCount) cartCount.innerText = totalCount;
      if (cartTotal) cartTotal.innerText = `$${totalPrice.toLocaleString('es-CO')} COP`;

      if (cartItems) {
        if (cart.length === 0) {
          cartItems.innerHTML = '<p style="text-align: center; color: #a8a29e; padding: 32px 0;">Tu carrito está vacío</p>';
        } else {
          cartItems.innerHTML = cart.map(p => `
            <div class="cart-item">
              <div class="cart-item-info">
                <h4>${p.title}</h4>
                <p>$${p.price.toLocaleString('es-CO')}</p>
              </div>
              <div class="qty-controls">
                <button onclick="changeQty(${p.id}, -1)" class="qty-btn">-</button>
                <span style="font-size: 13px; font-weight: bold;">${p.qty}</span>
                <button onclick="changeQty(${p.id}, 1)" class="qty-btn">+</button>
              </div>
            </div>
          `).join('');
        }
      }
    }

    function changeQty(id, delta) {
      const item = cart.find(p => p.id === id);
      if (item) {
        item.qty += delta;
        if (item.qty <= 0) cart = cart.filter(p => p.id !== id);
      }
      updateCartUI();
    }

    function toggleCart(forceOpen = false) {
      const modal = document.getElementById('cart-modal');
      if (!modal) return;
      if (forceOpen) modal.classList.add('active');
      else modal.classList.toggle('active');
    }

    function openImageModal(imgSrc, title) {
      const modal = document.getElementById('imgModal');
      const modalImg = document.getElementById('modalImg');
      modalImg.src = imgSrc;
      modal.classList.add('active');
    }

    function closeImageModal() {
      const modal = document.getElementById('imgModal');
      modal.classList.remove('active');
    }

    function checkoutWhatsApp() {
      if (cart.length === 0) {
        alert("Agrega al menos un producto al carrito para continuar.");
        return;
      }

      const phone = "573002869632";
      let message = "🌿 *NUEVO PEDIDO - AROMANTRIKA*\n\nHola, me gustaría solicitar los siguientes productos:\n\n";

      cart.forEach(p => {
        message += `• *${p.title}* (${p.presentacion}) x${p.qty} - $${(p.price * p.qty).toLocaleString('es-CO')}\n`;
      });

      const totalProductos = cart.reduce((acc, p) => acc + (p.price * p.qty), 0);
      
      message += `\n*Subtotal productos:* $${totalProductos.toLocaleString('es-CO')} COP\n`;
      message += `📦 *Envío:* Se calcula según ciudad y peso (Medellín/Área metropolitana o Nacional).\n`;
      message += `\nQuedo atento para confirmar el valor total del envío, la dirección de destino y coordinar el pago. ¡Muchas gracias!`;

      window.open(`https://wa.me/${phone}?text=${encodeURIComponent(message)}`, '_blank');
    }
