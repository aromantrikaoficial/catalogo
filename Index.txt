  <script>
    let cart = [];

    function addToCartDirect(title, price, presentacion) {
      const item = cart.find(p => p.title === title);
      if (item) { 
        item.qty++; 
      } else { 
        cart.push({ title, price, presentacion, qty: 1 }); 
      }
      updateCartUI();
      toggleCart(true);
    }

    function updateCartUI() {
      const cartItems = document.getElementById('cart-items');
      const cartCount = document.getElementById('cart-count');
      const cartCountFloat = document.getElementById('cart-count-float');
      const cartTotal = document.getElementById('cart-total');

      const totalCount = cart.reduce((acc, p) => acc + p.qty, 0);
      const totalPrice = cart.reduce((acc, p) => acc + (p.price * p.qty), 0);

      if (cartCount) cartCount.innerText = totalCount;
      if (cartCountFloat) cartCountFloat.innerText = totalCount;
      if (cartTotal) cartTotal.innerText = `$${totalPrice.toLocaleString('es-CO')} COP`;

      if (cartItems) {
        if (cart.length === 0) {
          cartItems.innerHTML = '<p style="text-align: center; color: #a8a29e; padding: 32px 0;">Tu carrito está vacío</p>';
        } else {
          cartItems.innerHTML = cart.map((p, index) => `
            <div class="cart-item">
              <div class="cart-item-info">
                <h4>${p.title}</h4>
                <p>$${p.price.toLocaleString('es-CO')} (${p.presentacion})</p>
              </div>
              <div class="qty-controls">
                <button onclick="changeQty(${index}, -1)" class="qty-btn">-</button>
                <span style="font-size: 13px; font-weight: bold;">${p.qty}</span>
                <button onclick="changeQty(${index}, 1)" class="qty-btn">+</button>
              </div>
            </div>
          `).join('');
        }
      }
    }

    function changeQty(index, delta) {
      cart[index].qty += delta;
      if (cart[index].qty <= 0) { 
        cart.splice(index, 1); 
      }
      updateCartUI();
    }

    function toggleCart(forceOpen = false) {
      const modal = document.getElementById('cart-modal');
      if (!modal) return;
      if (forceOpen) { 
        modal.classList.add('active'); 
      } else { 
        modal.classList.toggle('active'); 
      }
    }
